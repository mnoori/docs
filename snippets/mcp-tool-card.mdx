    );
  }
  const badges = {
    Onboarding:      "#00E6E2",
    Dispatch:        "#818CF8",
    Catalog:         "#22D3EE",

    Releases:        "#C084FC",
    Playlists:       "#4ADE80",
    "Smart Links":   "#FACC15",
    Storefront:      "#F59E0B",
    "Music Data":    "#FB923C",
    "AI Generation": "#F472B6",
    "Asset Upload":  "#A78BFA",
    Projects:        "#E879F9",
  };
  const color = badges[t.category] ?? "#A1A1AA";
  return (
    <div
      style={{
        padding: "16px",
        borderRadius: "12px",
        border: "1px solid rgba(255,255,255,0.10)",
        background: "rgba(255,255,255,0.03)",

          }}
        >
          {t.category}
        </span>
      </div>
      {t.callFirst && (
        <div style={{ fontSize: "11px", color: "#00E6E2", fontWeight: 600, marginBottom: "6px" }}>
          ★ Call this FIRST
        </div>
      )}
      <div style={{ fontSize: "14px", lineHeight: 1.5, color: "inherit" }}>{t.summary}</div>

};
